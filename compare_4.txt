"""Terminal User Interface for the File Aggregator.

Launch via:  aggt

Requires:
    pip install textual
"""

from __future__ import annotations

import sys
from pathlib import Path
from typing import ClassVar

from textual import on, work
from textual.app import App, ComposeResult
from textual.binding import Binding
from textual.containers import Horizontal, ScrollableContainer, Vertical
from textual.reactive import reactive
from textual.widgets import (
    Button,
    Checkbox,
    Footer,
    Header,
    Label,
    Log,
    Static,
)

# ---------------------------------------------------------------------------
# Resolve the project directory and import the core engine
# ---------------------------------------------------------------------------

_PROJECT_DIR = Path(__file__).resolve().parent
sys.path.insert(0, str(_PROJECT_DIR))

from aggregator import (           # noqa: E402
    aggregate_files,
    find_project_root,
    generate_tree,
    load_ignore_patterns,
    read_file_paths,
    should_ignore,
)

_FILES_TXT   = _PROJECT_DIR / "files.txt"
_ARENA_TXT   = _PROJECT_DIR / "arena.txt"
_STRUCTURE_TXT = _PROJECT_DIR / "structure.txt"


# ---------------------------------------------------------------------------
# CSS — kept minimal to conserve resources
# ---------------------------------------------------------------------------

_CSS = """
Screen {
    layout: vertical;
}

#body {
    layout: horizontal;
    height: 1fr;
}

/* ── Left panel: project tree ─────────────────────────────── */
#tree-panel {
    width: 40%;
    border: solid $primary;
    padding: 0 1;
}

#tree-title {
    background: $primary;
    color: $text;
    text-align: center;
    padding: 0 1;
}

#tree-scroll {
    height: 1fr;
}

/* ── Right panel: queue + log ─────────────────────────────── */
#right-panel {
    width: 60%;
    layout: vertical;
}

#queue-panel {
    height: 50%;
    border: solid $accent;
    padding: 0 1;
}

#queue-title {
    background: $accent;
    color: $text;
    text-align: center;
    padding: 0 1;
}

#queue-scroll {
    height: 1fr;
}

#log-panel {
    height: 50%;
    border: solid $surface;
    padding: 0 1;
}

/* ── Control bar ──────────────────────────────────────────── */
#controls {
    height: 3;
    layout: horizontal;
    align: center middle;
    padding: 0 1;
    background: $surface;
}

Button {
    margin: 0 1;
}

.status-ok  { color: $success; }
.status-err { color: $error;   }
"""


# ---------------------------------------------------------------------------
# Widgets
# ---------------------------------------------------------------------------


class TreeEntry(Checkbox):
    """A single filesystem entry displayed as a selectable checkbox.

    Attributes:
        file_path: Absolute path this entry represents.
    """

    DEFAULT_CSS = "TreeEntry { height: 1; }"

    def __init__(self, label: str, file_path: Path, value: bool = False) -> None:
        super().__init__(label, value=value)
        self.file_path = file_path


# ---------------------------------------------------------------------------
# Main application
# ---------------------------------------------------------------------------


class AggregatorTUI(App[None]):
    """Interactive TUI for browsing, selecting, and aggregating project files.

    Keyboard shortcuts
    ------------------
    r   Refresh the project tree.
    a   Run aggregation.
    c   Clear the current queue.
    q   Quit the application.
    """

    TITLE   = "File Aggregator"
    CSS     = _CSS

    BINDINGS: ClassVar[list[Binding]] = [
        Binding("r", "refresh",   "Refresh Tree",  show=True),
        Binding("a", "aggregate", "Aggregate",      show=True),
        Binding("c", "clear",     "Clear Queue",    show=True),
        Binding("q", "quit",      "Quit",           show=True),
    ]

    # Reactive counter drives the queue-title label automatically
    _queue_count: reactive[int] = reactive(0)

    # ------------------------------------------------------------------
    # Lifecycle
    # ------------------------------------------------------------------

    def on_mount(self) -> None:
        """Populate the tree and queue when the app first loads."""
        self._load_tree()
        self._load_queue()

    # ------------------------------------------------------------------
    # Layout
    # ------------------------------------------------------------------

    def compose(self) -> ComposeResult:
        yield Header()

        with Horizontal(id="body"):

            # ── Left: project tree ────────────────────────────────────
            with Vertical(id="tree-panel"):
                yield Static("📁  Project Tree  (click to queue)", id="tree-title")
                with ScrollableContainer(id="tree-scroll"):
                    pass  # populated by _load_tree()

            # ── Right: queue + log ────────────────────────────────────
            with Vertical(id="right-panel"):

                with Vertical(id="queue-panel"):
                    yield Static("📋  Queue  (0 files)", id="queue-title")
                    with ScrollableContainer(id="queue-scroll"):
                        pass  # populated by _load_queue()

                with Vertical(id="log-panel"):
                    yield Log(id="activity-log", auto_scroll=True)

        with Horizontal(id="controls"):
            yield Button("⟳  Refresh",   id="btn-refresh",   variant="default")
            yield Button("▶  Aggregate", id="btn-aggregate", variant="primary")
            yield Button("✕  Clear",     id="btn-clear",     variant="warning")
            yield Button("⏻  Quit",      id="btn-quit",      variant="error")

        yield Footer()

    # ------------------------------------------------------------------
    # Tree loading
    # ------------------------------------------------------------------

    def _load_tree(self) -> None:
        """Scan the project directory and populate the tree panel."""
        log = self.query_one("#activity-log", Log)
        scroll = self.query_one("#tree-scroll", ScrollableContainer)
        scroll.remove_children()

        # Detect root from CWD or the first queued path
        root: Path | None = None
        if _FILES_TXT.is_file():
            try:
                paths = read_file_paths(_FILES_TXT)
                if paths:
                    root = find_project_root(paths[0])
            except Exception:  # noqa: BLE001
                pass

        root = root or _PROJECT_DIR
        patterns = load_ignore_patterns(root)

        self._populate_tree(scroll, root, root, patterns)
        log.write_line(f"[tree] Loaded from: {root}")

    def _populate_tree(
        self,
        container: ScrollableContainer,
        dir_path: Path,
        root: Path,
        patterns: frozenset[str],
        depth: int = 0,
    ) -> None:
        """Recursively mount :class:`TreeEntry` widgets for *dir_path*.

        Args:
            container: The widget that receives new children.
            dir_path:  Directory currently being scanned.
            root:      Project root for ignore-pattern evaluation.
            patterns:  Active ignore patterns.
            depth:     Current indentation depth (0 = root level).
        """
        if depth > 6:   # shallow limit keeps the tree readable
            return

        try:
            items = sorted(
                dir_path.iterdir(),
                key=lambda p: (not p.is_dir(), p.name.lower()),
            )
        except PermissionError:
            return

        for item in items:
            if should_ignore(item, root, patterns):
                continue

            indent = "  " * depth
            if item.is_dir() and not item.is_symlink():
                container.mount(
                    Label(f"{indent}📁 {item.name}/", classes="tree-dir")
                )
                self._populate_tree(container, item, root, patterns, depth + 1)
            elif item.is_file():
                # Pre-check the box if already in the queue
                already_queued = self._path_in_queue(item)
                entry = TreeEntry(
                    f"{indent}  {item.name}",
                    file_path=item,
                    value=already_queued,
                )
                container.mount(entry)

    def _path_in_queue(self, path: Path) -> bool:
        """Return True if *path* already exists in files.txt."""
        if not _FILES_TXT.is_file():
            return False
        try:
            return path in read_file_paths(_FILES_TXT)
        except Exception:  # noqa: BLE001
            return False

    # ------------------------------------------------------------------
    # Queue loading
    # ------------------------------------------------------------------

    def _load_queue(self) -> None:
        """Refresh the queue panel from files.txt."""
        scroll  = self.query_one("#queue-scroll", ScrollableContainer)
        title   = self.query_one("#queue-title",  Static)
        log     = self.query_one("#activity-log", Log)

        scroll.remove_children()

        if not _FILES_TXT.is_file():
            self._queue_count = 0
            title.update("📋  Queue  (0 files)")
            return

        try:
            paths = read_file_paths(_FILES_TXT)
        except Exception as exc:  # noqa: BLE001
            log.write_line(f"[error] Could not read files.txt: {exc}")
            return

        for path in paths:
            label = str(path)
            scroll.mount(Label(label))

        self._queue_count = len(paths)
        title.update(f"📋  Queue  ({self._queue_count} files)")
        log.write_line(f"[queue] {self._queue_count} file(s) loaded.")

    # ------------------------------------------------------------------
    # Event handlers — checkboxes & buttons
    # ------------------------------------------------------------------

    @on(Checkbox.Changed)
    def handle_checkbox(self, event: Checkbox.Changed) -> None:
        """Add or remove a file from files.txt when its checkbox changes.

        Args:
            event: The change event carrying the new value.
        """
        if not isinstance(event.checkbox, TreeEntry):
            return

        entry = event.checkbox
        log   = self.query_one("#activity-log", Log)
        self._update_files_txt(entry.file_path, add=event.value)

        action = "Added" if event.value else "Removed"
        log.write_line(f"[queue] {action}: {entry.file_path.name}")
        self._load_queue()

    @on(Button.Pressed, "#btn-refresh")
    def handle_refresh(self) -> None:
        self.action_refresh()

    @on(Button.Pressed, "#btn-aggregate")
    def handle_aggregate(self) -> None:
        self.action_aggregate()

    @on(Button.Pressed, "#btn-clear")
    def handle_clear(self) -> None:
        self.action_clear()

    @on(Button.Pressed, "#btn-quit")
    def handle_quit(self) -> None:
        self.action_quit()

    # ------------------------------------------------------------------
    # Actions (also reachable via key bindings)
    # ------------------------------------------------------------------

    def action_refresh(self) -> None:
        """Re-scan the project tree and reload the queue."""
        self._load_tree()
        self._load_queue()

    @work(thread=True)
    def action_aggregate(self) -> None:
        """Run core aggregation in a background thread (non-blocking).

        Posts status messages back to the UI log upon completion.
        """
        log = self.query_one("#activity-log", Log)
        self.call_from_thread(log.write_line, "[run] Starting aggregation…")

        try:
            paths = read_file_paths(_FILES_TXT)
            if not paths:
                self.call_from_thread(
                    log.write_line, "[warn] files.txt is empty — nothing to do."
                )
                return

            root     = find_project_root(paths[0])
            patterns = load_ignore_patterns(root)

            # Project tree
            if root:
                tree_lines = [f"Project Root: {root.name}/"] + generate_tree(
                    root, root, patterns
                )
                _STRUCTURE_TXT.write_text("\n".join(tree_lines), encoding="utf-8")
                self.call_from_thread(
                    log.write_line, f"[ok] structure.txt written ({root.name})"
                )

            # File aggregation — redirect stderr to log
            aggregate_files(paths, _ARENA_TXT, root)
            self.call_from_thread(
                log.write_line,
                f"[ok] arena.txt written ({len(paths)} file(s)).",
            )

        except FileNotFoundError:
            self.call_from_thread(
                log.write_line, "[error] files.txt not found — add files first."
            )
        except Exception as exc:  # noqa: BLE001
            self.call_from_thread(log.write_line, f"[error] {exc}")

    def action_clear(self) -> None:
        """Empty files.txt and uncheck all tree entries."""
        _FILES_TXT.write_text("", encoding="utf-8")

        for entry in self.query(TreeEntry):
            entry.value = False             # triggers no file-write (guard below)

        log = self.query_one("#activity-log", Log)
        log.write_line("[queue] Cleared.")
        self._load_queue()

    # ------------------------------------------------------------------
    # files.txt mutation helpers
    # ------------------------------------------------------------------

    def _update_files_txt(self, path: Path, *, add: bool) -> None:
        """Persist a path addition or removal to files.txt.

        Args:
            path: The file path to add or remove.
            add:  ``True`` to add, ``False`` to remove.
        """
        existing: list[Path] = []
        if _FILES_TXT.is_file():
            try:
                existing = read_file_paths(_FILES_TXT)
            except Exception:  # noqa: BLE001
                pass

        if add:
            if path not in existing:
                existing.append(path)
        else:
            existing = [p for p in existing if p != path]

        _FILES_TXT.write_text(
            "\n".join(str(p) for p in existing) + ("\n" if existing else ""),
            encoding="utf-8",
        )


# ---------------------------------------------------------------------------
# Entry point
# ---------------------------------------------------------------------------

def main() -> None:
    """Launch the TUI application."""
    AggregatorTUI().run()


if __name__ == "__main__":
    main()

    # ------------------------------------------------------------------
    # Project tree population
    # ------------------------------------------------------------------

    def _refresh_tree(self) -> None:
        """Clear and re-populate the Treeview from the project directory."""
        self._tree.delete(*self._tree.get_children())
        self._checked.clear()

        root: Optional[Path] = None
        if _FILES_TXT.is_file():
            try:
                paths = read_file_paths(_FILES_TXT)
                if paths:
                    root = find_project_root(paths[0])
            except Exception:  # noqa: BLE001
                pass

        root = root or _PROJECT_DIR
        patterns = load_ignore_patterns(root)
        queued = self._current_queue_paths()

        self._populate_tree("", root, root, patterns, queued)
        self._log_write(f"Tree loaded from: {root}", tag="info")

    def _populate_tree(
        self,
        parent_iid: str,
        dir_path: Path,
        root: Path,
        patterns: frozenset[str],
        queued: set[Path],
        depth: int = 0,
    ) -> None:
        """Recursively insert items into the Treeview.

        Args:
            parent_iid: Treeview parent node identifier.
            dir_path:   Directory being scanned.
            root:       Project root for ignore evaluation.
            patterns:   Active ignore patterns.
            queued:     Paths already present in the queue.
            depth:      Current recursion depth (display limit: 8).
        """
        if depth > 8:
            return

        try:
            items = sorted(
                dir_path.iterdir(),
                key=lambda p: (not p.is_dir(), p.name.lower()),
            )
        except PermissionError:
            return

        for item in items:
            if should_ignore(item, root, patterns):
                continue

            if item.is_dir() and not item.is_symlink():
                iid = self._tree.insert(
                    parent_iid, "end",
                    text=f"📁 {item.name}",
                    open=False,
                    values=[str(item)],
                )
                self._populate_tree(iid, item, root, patterns, queued, depth + 1)

            elif item.is_file():
                marker = "✔ " if item in queued else "  "
                iid = self._tree.insert(
                    parent_iid, "end",
                    text=f"{marker}{item.name}",
                    values=[str(item)],
                )

    # ------------------------------------------------------------------
    # Queue management
    # ------------------------------------------------------------------

    def _current_queue_paths(self) -> set[Path]:
        """Return the set of paths currently recorded in files.txt."""
        if not _FILES_TXT.is_file():

```python
"""Core library for the File Aggregator tool.

All three interfaces (agg, aggt, aggg) import from this module.
"""

import sys
import fnmatch
from pathlib import Path
from typing import List, Optional, Set, FrozenSet

_ROOT_MARKERS: FrozenSet[str] = frozenset({
    "package.json", ".git", "requirements.txt", "pyproject.toml", "src"
})

_DEFAULT_IGNORE: FrozenSet[str] = frozenset({
    ".git", "node_modules", "dist", "__pycache__", ".next",
    ".index_ignore", "*.pyc", ".DS_Store", "build", ".venv"
})

_MAX_TREE_DEPTH = 20


def find_project_root(path: Path) -> Optional[Path]:
    """Find nearest project root by looking for known marker files/directories.

    Args:
        path: Any path inside the project.

    Returns:
        The project root Path or None if no marker is found.
    """
    current = path.resolve().parent
    while True:
        if any((current / marker).exists() for marker in _ROOT_MARKERS):
            return current
        parent = current.parent
        if parent == current:
            return None
        current = parent


def get_display_path(path: Path, root: Optional[Path]) -> str:
    """Return clean relative path, falling back intelligently.

    Resolution order: root → cwd → absolute (as_posix).
    """
    abs_path = path.resolve()
    for anchor in (root, Path.cwd()):
        if anchor is None:
            continue
        try:
            return abs_path.relative_to(anchor).as_posix()
        except ValueError:
            continue
    return abs_path.as_posix()


def load_ignore_patterns(root: Optional[Path]) -> FrozenSet[str]:
    """Load .index_ignore (if present) and merge with defaults.

    Args:
        root: Project root. Falls back to cwd when None.

    Returns:
        Immutable set of glob patterns to ignore.
    """
    patterns: Set[str] = set(_DEFAULT_IGNORE)
    search_dir = root or Path.cwd()
    ignore_file = search_dir / ".index_ignore"

    if ignore_file.is_file():
        try:
            with ignore_file.open("r", encoding="utf-8") as f:
                for line in f:
                    stripped = line.strip()
                    if stripped and not stripped.startswith("#"):
                        patterns.add(stripped)
        except OSError as exc:
            print(f"Warning: Could not read .index_ignore: {exc}", file=sys.stderr)

    return frozenset(patterns)


def should_ignore(path: Path, root: Path, patterns: FrozenSet[str]) -> bool:
    """Check if a path should be ignored."""
    try:
        rel = path.relative_to(root)
    except ValueError:
        return False

    rel_posix = rel.as_posix()
    parts = rel.parts

    return any(
        fnmatch.fnmatch(rel_posix, pat) or
        any(fnmatch.fnmatch(part, pat) for part in parts)
        for pat in patterns
    )


def generate_tree(
    dir_path: Path,
    root: Path,
    patterns: FrozenSet[str],
    prefix: str = "",
    depth: int = 0,
) -> List[str]:
    """Generate visual directory tree (used by CLI and both UIs)."""
    if depth > _MAX_TREE_DEPTH:
        return [f"{prefix}... (max depth reached)"]

    tree: List[str] = []
    try:
        items = sorted(
            dir_path.iterdir(),
            key=lambda x: (not x.is_dir(), x.name.lower())
        )
        items = [item for item in items if not should_ignore(item, root, patterns)]
    except PermissionError:
        return [f"{prefix}[Permission Denied]"]
    except OSError as exc:
        return [f"{prefix}[Error: {exc}]"]

    for i, item in enumerate(items):
        is_last = i == len(items) - 1
        connector = "└── " if is_last else "├── "
        suffix = "/" if item.is_dir() else ""
        tree.append(f"{prefix}{connector}{item.name}{suffix}")

        if item.is_dir() and not item.is_symlink():
            new_prefix = prefix + ("    " if is_last else "│   ")
            tree.extend(generate_tree(item, root, patterns, new_prefix, depth + 1))

    return tree


def read_file_paths(source_file: Path) -> List[Path]:
    """Read list of files from files.txt (used by original CLI)."""
    if not source_file.is_file():
        raise FileNotFoundError(f"Source file not found: {source_file}")

    paths: List[Path] = []
    with source_file.open("r", encoding="utf-8") as f:
        for line in f:
            stripped = line.strip()
            if stripped:
                paths.append(Path(stripped))
    return paths


def aggregate_files(paths: List[Path], output_file: Path, root: Optional[Path]) -> None:
    """Aggregate selected files into arena.txt with clean headers.

    Reads content *before* writing header to prevent orphaned headers.
    """
    with output_file.open("w", encoding="utf-8") as out:
        for path in paths:
            try:
                if not path.is_file():
                    print(f"ERROR: Not a file: {path}", file=sys.stderr)
                    continue

                content = path.read_text(encoding="utf-8")
                display = get_display_path(path, root)

                out.write(f"# --- FILE: {display} ---\n")
                out.write(content)
                if not content.endswith("\n"):
                    out.write("\n")
                out.write("\n")
            except PermissionError as e:
                print(f"ERROR: Permission denied reading {path}: {e}", file=sys.stderr)
            except UnicodeDecodeError as e:
                print(f"ERROR: Encoding error in {path}: {e}", file=sys.stderr)
            except OSError as e:
                print(f"ERROR: OS error with {path}: {e}", file=sys.stderr)


def write_selected_paths(paths: List[Path], target: Path = Path("files.txt")) -> None:
    """Save currently selected files to files.txt for CLI compatibility."""
    target.write_text("\n".join(str(p.resolve()) for p in paths), encoding="utf-8")

# aggregator.py
import os
import subprocess
from pathlib import Path
from typing import List, Dict, Optional, Tuple
import fnmatch

# Type aliases for clarity
FileTree = Dict[str, any]

def find_project_root(path: Optional[Path] = None) -> Path:
    """Find the project root by looking for .git or falling back to cwd."""
    current = path or Path.cwd()
    while current != current.parent:
        if (current / ".git").exists():
            return current
        current = current.parent
    return Path.cwd()

def load_ignore_patterns(root: Path) -> List[str]:
    """Load patterns from .gitignore or .aggignore."""
    ignore_file = root / ".gitignore"
    patterns = [".git", "__pycache__", "*.pyc", "node_modules", ".env"]
    
    if ignore_file.exists():
        with open(ignore_file, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if line and not line.startswith("#"):
                    patterns.append(line)
    return patterns

def should_ignore(path: Path, root: Path, patterns: List[str]) -> bool:
    """Check if a path matches any ignore pattern."""
    rel_path = path.relative_to(root)
    rel_str = rel_path.as_posix()
    
    for pattern in patterns:
        # Handle directory patterns (ending with /)
        if pattern.endswith("/"):
            if path.is_dir() and fnmatch.fnmatch(rel_str, pattern[:-1]):
                return True
            if rel_str.startswith(pattern):
                return True
        # Handle standard patterns
        if fnmatch.fnmatch(rel_str, pattern) or fnmatch.fnmatch(path.name, pattern):
            return True
            
    return False

def generate_tree(root: Path, patterns: List[str]) -> Tuple[FileTree, List[Path]]:
    """
    Generate a nested dictionary representing the file tree.
    Returns the tree structure and a flat list of all valid file paths.
    """
    tree = {}
    all_files = []
    
    for path in root.rglob("*"):
        if should_ignore(path, root, patterns):
            continue
            
        if path.is_file():
            all_files.append(path)
            parts = path.relative_to(root).parts
            current_level = tree
            for part in parts[:-1]:
                if part not in current_level:
                    current_level[part] = {}
                current_level = current_level[part]
            current_level[parts[-1]] = None # File marker

    return tree, all_files

def aggregate_files(files: List[Path], root: Path) -> str:
    """Combine file contents into a single string with headers."""
    output = []
    for file in files:
        rel_path = file.relative_to(root)
        try:
            content = file.read_text(encoding="utf-8", errors="ignore")
            output.append(f"--- File: {rel_path} ---\n{content}\n")
        except Exception as e:
            output.append(f"--- Error reading {rel_path}: {e} ---\n")
    return "\n".join(output)

def save_output(content: str, root: Path, filename: str = "arena.txt") -> Path:
    """Save content to the specified file in the project root."""
    out_path = root / filename
    out_path.write_text(content, encoding="utf-8")
    return out_path
```

        yield Header()
        yield Container(
            Vertical(
                Label(f"Project: {self.root.name}", classes="header"),
                Button("Aggregate Selected", variant="primary", id="btn_agg"),
                Button("Select All", id="btn_all"),
                Button("Clear Selection", id="btn_clear"),
                Label("File Browser:"),
                DirectoryTree(self.root, id="tree"),
                id="sidebar"
            ),
            Vertical(
                Label("Status Log"),
                Static(id="log"),
                id="main"
            )
        )
        yield Footer()

    def on_directory_tree_file_selected(self, event: DirectoryTree.FileSelected) -> None:
        """Handle file selection toggle."""
        path = event.path
        if path in self.selected_files:
            self.selected_files.remove(path)
            self.log(f"Deselected: {path.name}")
        else:
            self.selected_files.add(path)
            self.log(f"Selected: {path.name}")

    def on_button_pressed(self, event: Button.Pressed) -> None:
        btn_id = event.button.id
        if btn_id == "btn_agg":
            self.action_aggregate()
        elif btn_id == "btn_all":
            self.selected_files = set(self.all_files)
            self.log(f"Selected {len(self.selected_files)} files.")
        elif btn_id == "btn_clear":
            self.selected_files.clear()
            self.log("Selection cleared.")

    def action_aggregate(self) -> None:
        if not self.selected_files:
            self.log("No files selected!")
            return
        
        self.log(f"Aggregating {len(self.selected_files)} files...")
        # Run aggregation in thread to avoid blocking UI
        def run_agg():
            content = core.aggregate_files(list(self.selected_files), self.root)
            out_path = core.save_output(content, self.root)
            self.log(f"Saved to {out_path.name}")
        
        # Simple non-async delegation (in prod, use self.run_worker)
        import threading
        t = threading.Thread(target=run_agg)
        t.start()
        t.join()
        self.log("Done.")

    def log(self, message: str) -> None:
        log_widget = self.query_one("#log", Static)
        current_text = log_widget.renderable or ""
        log_widget.update(f"{current_text}\n> {message}")

if __name__ == "__main__":
    app = AggregatorTUI()
    app.run()

        ttk.Label(main_frame, textvariable=self.status_var, relief=tk.SUNKEN).pack(fill=tk.X, pady=(10, 0), side=tk.BOTTOM)

        """Recursively populate the Tkinter tree."""
        self.tree.delete(*self.tree.get_children())
        self._insert_node("", self.root_path, self.tree_data)

    def _insert_node(self, parent, path: Path, node_data: dict):
        """Helper to insert nodes recursively."""
        text = path.name if path != self.root_path else self.root_path.name
        node_id = self.tree.insert(parent, "end", text=text, open=False)
        
        if isinstance(node_data, dict):
            # It's a directory
            for name, child_data in sorted(node_data.items()):
                child_path = path / name
                self._insert_node(node_id, child_path, child_data)
        else:
            # It's a file
            pass

    def _on_double_click(self, event):
        """Toggle selection on double click."""
        item_id = self.tree.focus()
        item_text = self.tree.item(item_id, "text")
        # This is a simplified logic: in a real app, we'd track full paths better in the tree items
        # For brevity, we just show how to interact
        self.status_var.set(f"Selected item: {item_text}")

        """Run aggregation in a separate thread."""
        # For this simple GUI, we'll just grab all files found in core generation
        # Ideally, we'd sync the Treeview selection with the file list
        self.selected_files = set(self.all_files) # Simplified: Aggregate all valid files
        
        self.status_var.set("Processing...")
        threading.Thread(target=self._run_aggregation, daemon=True).start()

        self.status_var.set("Selection cleared")

import os
import subprocess
import sys
from contextlib import contextmanager
from dataclasses import dataclass
from pathlib import Path

FILES_TXT_NAME = "files.txt"


@contextmanager
def pushd(path: Path) -> Iterable[None]:
    """Temporarily change working directory."""
    old = Path.cwd()
    os.chdir(path)
    try:
        yield
    finally:
        os.chdir(old)


@dataclass(frozen=True)
class AggregatorEnv:
    project_root: Path
    aggregator_py: Path


def _find_project_root_fallback(start: Path) -> Path:
    """Fallback project-root detection (used if aggregator.find_project_root is unavailable)."""
    cur = start.resolve()
    for parent in (cur, *cur.parents):
        if (parent / ".git").exists():
            return parent
        if (parent / "pyproject.toml").exists():
            return parent
    return cur


def load_aggregator_env() -> AggregatorEnv:
    """Locate aggregator.py and project root."""
    here = Path(__file__).resolve().parent
    aggregator_py = here / "aggregator.py"
    if not aggregator_py.exists():
        raise FileNotFoundError(f"Expected aggregator.py next to this file: {aggregator_py}")

    # Prefer aggregator.find_project_root if present, else fallback
    project_root: Path
    sys.path.insert(0, str(here))
    try:
        import aggregator  # type: ignore

        if hasattr(aggregator, "find_project_root"):
            project_root = Path(aggregator.find_project_root())  # type: ignore[arg-type]
        else:
            project_root = _find_project_root_fallback(Path.cwd())
    finally:
        if sys.path and sys.path[0] == str(here):
            sys.path.pop(0)

    return AggregatorEnv(project_root=project_root.resolve(), aggregator_py=aggregator_py.resolve())


def write_files_txt(project_root: Path, rel_files: Iterable[str]) -> Path:
    """Write files.txt in project root."""
    path = project_root / FILES_TXT_NAME
    content = "\n".join(rel_files).strip() + ("\n" if rel_files else "")
    path.write_text(content, encoding="utf-8")
    return path


def read_files_txt(project_root: Path) -> list[str]:
    """Read files.txt from project root."""
    path = project_root / FILES_TXT_NAME
    if not path.exists():
        return []
    lines = [ln.strip() for ln in path.read_text(encoding="utf-8").splitlines()]
    return [ln for ln in lines if ln and not ln.startswith("#")]


def run_aggregator(env: AggregatorEnv) -> None:
    """
    Run aggregator.py.

    Tries to call aggregator.main() if available (no subprocess).
    Falls back to subprocess otherwise.
    """
    here = env.aggregator_py.parent
    sys.path.insert(0, str(here))
    try:
        import aggregator  # type: ignore

        if hasattr(aggregator, "main"):
            with pushd(env.project_root):
                aggregator.main()  # type: ignore[misc]
            return
    except Exception:
        # If import/main fails, we fall back to subprocess
        pass
    finally:
        if sys.path and sys.path[0] == str(here):
            sys.path.pop(0)

    with pushd(env.project_root):
        subprocess.run([sys.executable, str(env.aggregator_py)], check=True)


class ProjectTree(Tree[Path]):
    """Lazy-loading file tree."""

    def __init__(self, root: Path) -> None:
        super().__init__(str(root), root)
        self.show_root = True
        self._add_placeholder(self.root)

    @staticmethod
    def _add_placeholder(node: TreeNode[Path]) -> None:
        # Ensure the node is expandable in the UI.
        if len(node.children) == 0:
            node.add("…", node.data)

    def populate_node(self, node: TreeNode[Path]) -> None:
        path = node.data
        if not path.is_dir():
            return

        # If placeholder exists, clear children and populate real entries.
        node.remove_children()

        try:
            entries = sorted(
                path.iterdir(),
                key=lambda p: (p.is_file(), p.name.lower()),
            )
        except (PermissionError, FileNotFoundError):
            return

        for entry in entries:
            # Keep this lightweight: skip very common heavy/noisy dirs by default.
            if entry.name in {".git", ".venv", "__pycache__", "node_modules"}:
                continue

            child = node.add(f"{entry.name}{'/' if entry.is_dir() else ''}", entry)
            if entry.is_dir():
                self._add_placeholder(child)

    def on_tree_node_expanded(self, event: Tree.NodeExpanded[Path]) -> None:
        self.tree.populate_node(event.node)

    def on_tree_node_selected(self, event: Tree.NodeSelected[Path]) -> None:
        self._toggle_path(event.node.data)

    def action_toggle(self) -> None:
        node = self.tree.cursor_node
        if node is not None:
            self._toggle_path(node.data)

    def action_clear(self) -> None:
        self.selected.clear()
        self._refresh_selected_list()
        self.log.write("Selection cleared.")

    def action_refresh(self) -> None:
        self.tree.root.remove_children()
        self.tree._add_placeholder(self.tree.root)
        self.tree.root.expand()
        self.tree.populate_node(self.tree.root)
        self.log.write("Tree refreshed.")

    def action_save(self) -> None:
        rels = [str(p.relative_to(self.env.project_root)) for p in sorted(self.selected)]
        path = write_files_txt(self.env.project_root, rels)
        self.log.write(f"Saved {len(rels)} file(s) to {path}")

    def action_load(self) -> None:
        rels = read_files_txt(self.env.project_root)
        selected: set[Path] = set()
        for rel in rels:
            p = (self.env.project_root / rel).resolve()
            if p.exists() and p.is_file():
                selected.add(p)
        self.selected = selected
        self._refresh_selected_list()
        self.log.write(f"Loaded {len(self.selected)} file(s) from files.txt")

    def action_aggregate(self) -> None:
        # Ensure aggregator sees the same selection as CLI behavior.
        self.action_save()
        try:
            run_aggregator(self.env)
        except subprocess.CalledProcessError as exc:
            self.log.write(f"Aggregation failed (exit={exc.returncode}).")
        except Exception as exc:  # noqa: BLE001
            self.log.write(f"Aggregation failed: {exc!r}")
        else:
            self.log.write("Aggregation complete.")

    def on_button_pressed(self, event: Button.Pressed) -> None:
        match event.button.id:
            case "btn_aggregate":
                self.action_aggregate()
            case "btn_save":
                self.action_save()
            case "btn_load":
                self.action_load()
            case "btn_clear":
                self.action_clear()
            case "btn_refresh":
                self.action_refresh()


if __name__ == "__main__":
    AggregatorTUI().run()

## 3) Create `aggregator_gui.py` (Tkinter, command: `aggg`)

Also save next to `aggregator.py`:

```python
from __future__ import annotations

import os
import subprocess
import sys
from contextlib import contextmanager
from dataclasses import dataclass
from pathlib import Path
from tkinter import END, BOTH, LEFT, RIGHT, SINGLE, TOP, VERTICAL, Y, Tk, messagebox
from tkinter import ttk
from typing import Iterable


FILES_TXT_NAME = "files.txt"


@contextmanager
def pushd(path: Path) -> Iterable[None]:
    old = Path.cwd()
    os.chdir(path)
    try:
        yield
    finally:
        os.chdir(old)


@dataclass(frozen=True)
class AggregatorEnv:
    project_root: Path
    aggregator_py: Path


def _find_project_root_fallback(start: Path) -> Path:
    cur = start.resolve()
    for parent in (cur, *cur.parents):
        if (parent / ".git").exists():
            return parent
        if (parent / "pyproject.toml").exists():
            return parent
    return cur


def load_aggregator_env() -> AggregatorEnv:
    here = Path(__file__).resolve().parent
    aggregator_py = here / "aggregator.py"
    if not aggregator_py.exists():
        raise FileNotFoundError(f"Expected aggregator.py next to this file: {aggregator_py}")

    project_root: Path
    sys.path.insert(0, str(here))
    try:
        import aggregator  # type: ignore

        if hasattr(aggregator, "find_project_root"):
            project_root = Path(aggregator.find_project_root())  # type: ignore[arg-type]
        else:
            project_root = _find_project_root_fallback(Path.cwd())
    finally:
        if sys.path and sys.path[0] == str(here):
            sys.path.pop(0)

    return AggregatorEnv(project_root=project_root.resolve(), aggregator_py=aggregator_py.resolve())


def write_files_txt(project_root: Path, rel_files: Iterable[str]) -> Path:
    path = project_root / FILES_TXT_NAME
    content = "\n".join(rel_files).strip() + ("\n" if rel_files else "")
    path.write_text(content, encoding="utf-8")
    return path


def read_files_txt(project_root: Path) -> list[str]:
    path = project_root / FILES_TXT_NAME
    if not path.exists():
        return []
    lines = [ln.strip() for ln in path.read_text(encoding="utf-8").splitlines()]
    return [ln for ln in lines if ln and not ln.startswith("#")]


def run_aggregator(env: AggregatorEnv) -> None:
    here = env.aggregator_py.parent
    sys.path.insert(0, str(here))
    try:
        import aggregator  # type: ignore

        if hasattr(aggregator, "main"):
            with pushd(env.project_root):
                aggregator.main()  # type: ignore[misc]
            return
    except Exception:
        pass
    finally:
        if sys.path and sys.path[0] == str(here):
            sys.path.pop(0)

    with pushd(env.project_root):
        subprocess.run([sys.executable, str(env.aggregator_py)], check=True)


class AggregatorGUI:
    def __init__(self) -> None:
        self.env = load_aggregator_env()
        self.selected: set[Path] = set()

        self.root = Tk()
        self.root.title("File Aggregator (aggg)")
        self.root.geometry("1000x650")

        self._build_ui()
        self._load_tree_root()

    def _build_ui(self) -> None:
        outer = ttk.Panedwindow(self.root, orient="horizontal")
        outer.pack(fill=BOTH, expand=True)

        left = ttk.Frame(outer)
        right = ttk.Frame(outer)
        outer.add(left, weight=3)
        outer.add(right, weight=2)

        ttk.Label(left, text=f"Project: {self.env.project_root}").pack(anchor="w")

        self.tree = ttk.Treeview(left, selectmode=SINGLE)
        ysb = ttk.Scrollbar(left, orient=VERTICAL, command=self.tree.yview)
        self.tree.configure(yscroll=ysb.set)
        self.tree.pack(side=LEFT, fill=BOTH, expand=True)
        ysb.pack(side=RIGHT, fill=Y)

        self.tree.bind("<<TreeviewOpen>>", self._on_open_node)
        self.tree.bind("<Double-1>", self._on_double_click)

        ttk.Label(right, text="Selected (relative paths):").pack(anchor="w")
        self.selected_list = ttk.Treeview(right, columns=("path",), show="headings", height=20)
        self.selected_list.heading("path", text="path")
        self.selected_list.pack(fill=BOTH, expand=False)

        controls = ttk.Frame(right)
        controls.pack(fill=BOTH, expand=False)

        ttk.Button(controls, text="Aggregate", command=self.aggregate).pack(fill="x", pady=2)
        ttk.Button(controls, text="Save", command=self.save).pack(fill="x", pady=2)
        ttk.Button(controls, text="Load", command=self.load).pack(fill="x", pady=2)
        ttk.Button(controls, text="Clear", command=self.clear).pack(fill="x", pady=2)

    def _skip_name(self, name: str) -> bool:
        return name in {".git", ".venv", "__pycache__", "node_modules"}

    def _insert_children(self, parent_id: str, path: Path) -> None:
        # Remove dummy placeholder if present
        children = self.tree.get_children(parent_id)
        if len(children) == 1 and self.tree.item(children[0], "text") == "…":
            self.tree.delete(children[0])

        try:
            entries = sorted(path.iterdir(), key=lambda p: (p.is_file(), p.name.lower()))
        except (PermissionError, FileNotFoundError):
            return

        for entry in entries:
            if self._skip_name(entry.name):
                continue
            node_id = self.tree.insert(parent_id, END, text=entry.name, values=(str(entry),))
            if entry.is_dir():
                # Add dummy child to make it expandable
                self.tree.insert(node_id, END, text="…")

    def _load_tree_root(self) -> None:
        root_id = self.tree.insert("", END, text=str(self.env.project_root), values=(str(self.env.project_root),))
        self.tree.insert(root_id, END, text="…")
        self.tree.item(root_id, open=True)
        self._insert_children(root_id, self.env.project_root)

    def _on_open_node(self, _event: object) -> None:
        item_id = self.tree.focus()
        values = self.tree.item(item_id, "values")
        if not values:
            return
        path = Path(values[0])
        if path.is_dir():
            self._insert_children(item_id, path)

    def _refresh_selected_view(self) -> None:
        for row in self.selected_list.get_children(""):
            self.selected_list.delete(row)

        for p in sorted(self.selected):
            rel = str(p.relative_to(self.env.project_root))
            self.selected_list.insert("", END, values=(rel,))

    def _on_double_click(self, _event: object) -> None:
        item_id = self.tree.focus()
        values = self.tree.item(item_id, "values")
        if not values:
            return
        path = Path(values[0])
        if path.is_file():
            if path in self.selected:
                self.selected.remove(path)
            else:
                self.selected.add(path)
            self._refresh_selected_view()

    def save(self) -> None:
        rels = [str(p.relative_to(self.env.project_root)) for p in sorted(self.selected)]
        path = write_files_txt(self.env.project_root, rels)
        messagebox.showinfo("Saved", f"Saved {len(rels)} file(s) to:\n{path}")

    def load(self) -> None:
        rels = read_files_txt(self.env.project_root)
        selected: set[Path] = set()
        for rel in rels:
            p = (self.env.project_root / rel).resolve()
            if p.exists() and p.is_file():
                selected.add(p)
        self.selected = selected
        self._refresh_selected_view()
        messagebox.showinfo("Loaded", f"Loaded {len(self.selected)} file(s) from files.txt")

    def clear(self) -> None:
        self.selected.clear()
        self._refresh_selected_view()

    def aggregate(self) -> None:
        self.save()
        try:
            run_aggregator(self.env)
        except subprocess.CalledProcessError as exc:
            messagebox.showerror("Aggregation failed", f"Exit code: {exc.returncode}")
        except Exception as exc:  # noqa: BLE001
            messagebox.showerror("Aggregation failed", repr(exc))
        else:
            messagebox.showinfo("Done", "Aggregation complete.")

    def run(self) -> None:
        self.root.mainloop()


if __name__ == "__main__":
    AggregatorGUI().run()
```
