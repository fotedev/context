### 1. Core Logic (`aggregator.py`)

This file serves as the engine. It is refactored to be **importable** (for the GUI/TUI) while maintaining its **CLI functionality**.

```python
"""Core aggregation logic.
Designed to be imported by UIs or run directly via CLI.
"""

from __future__ import annotations

import sys
import fnmatch
from pathlib import Path
from typing import List, Optional, Set, FrozenSet

# Constants
ROOT_MARKERS: FrozenSet[str] = frozenset(
    {".git", "pyproject.toml", "setup.py", "requirements.txt", "package.json"}
)
DEFAULT_IGNORE: FrozenSet[str] = frozenset(
    {".git", "__pycache__", "node_modules", ".venv", "dist", "build", "*.pyc"}
)
FILES_TXT = "files.txt"
ARENA_TXT = "arena.txt"
STRUCTURE_TXT = "structure.txt"


def find_project_root(start_path: Path | str = ".") -> Path:
    """Find the project root by traversing upwards."""
    current = Path(start_path).resolve()
    while current != current.parent:
        if any((current / marker).exists() for marker in ROOT_MARKERS):
            return current
        current = current.parent
    return Path.cwd()


def load_ignore_patterns(root: Path) -> Set[str]:
    """Load ignore patterns from .gitignore or .index_ignore."""
    patterns = set(DEFAULT_IGNORE)
    ignore_file = root / ".gitignore" if (root / ".gitignore").exists() else root / ".index_ignore"
    
    if ignore_file.exists():
        try:
            with ignore_file.open("r", encoding="utf-8") as f:
                for line in f:
                    line = line.strip()
                    if line and not line.startswith("#"):
                        patterns.add(line)
        except OSError:
            pass
    return patterns


def should_ignore(path: Path, root: Path, patterns: Set[str]) -> bool:
    """Check if a path matches any ignore pattern."""
    try:
        rel_path = path.relative_to(root)
        rel_str = rel_path.as_posix()
    except ValueError:
        return False

    for pattern in patterns:
        # Directory pattern matching
        if pattern.endswith("/") and path.is_dir():
            if fnmatch.fnmatch(rel_str, pattern[:-1]) or fnmatch.fnmatch(path.name, pattern[:-1]):
                return True
        # Standard glob matching
        if fnmatch.fnmatch(rel_str, pattern) or fnmatch.fnmatch(path.name, pattern):
            return True
    return False


def generate_tree_lines(
    dir_path: Path, root: Path, patterns: Set[str], prefix: str = ""
) -> List[str]:
    """Recursively generate a list of strings for the project tree."""
    if not dir_path.is_dir():
        return []

    contents = list(dir_path.iterdir())
    contents = [p for p in contents if not should_ignore(p, root, patterns)]
    contents.sort(key=lambda p: (p.is_file(), p.name.lower()))

    lines = []
    for i, path in enumerate(contents):
        connector = "└── " if i == len(contents) - 1 else "├── "
        lines.append(f"{prefix}{connector}{path.name}")
        
        if path.is_dir():
            extension = "    " if i == len(contents) - 1 else "│   "
            lines.extend(generate_tree_lines(path, root, patterns, prefix + extension))
    return lines


def aggregate_files(file_paths: List[Path], root: Path, output_path: Path) -> int:
    """Write selected file contents to arena.txt."""
    count = 0
    with output_path.open("w", encoding="utf-8") as out:
        for path in file_paths:
            try:
                rel_path = path.relative_to(root)
                content = path.read_text(encoding="utf-8")
                out.write(f"# --- FILE: {rel_path} ---\n")
                out.write(content)
                if not content.endswith("\n"):
                    out.write("\n")
                out.write("\n")  # Extra newline between files
                count += 1
            except Exception as e:
                print(f"Error reading {path}: {e}", file=sys.stderr)
    return count


def read_file_paths(source_file: Path) -> List[Path]:
    """Read list of paths from a file (one per line)."""
    if not source_file.exists():
        return []
    paths = []
    with source_file.open("r", encoding="utf-8") as f:
        for line in f:
            p = line.strip()
            if p and not p.startswith("#"):
                paths.append(Path(p))
    return paths


def main() -> None:
    """CLI Entry Point."""
    root = find_project_root()
    files_txt = root / FILES_TXT
    arena_txt = root / ARENA_TXT
    structure_txt = root / STRUCTURE_TXT

    print(f"Project Root: {root}")
    
    # 1. Generate Structure
    patterns = load_ignore_patterns(root)
    tree_lines = [f"{root.name}/"] + generate_tree_lines(root, root, patterns)
    structure_txt.write_text("\n".join(tree_lines), encoding="utf-8")
    print(f"Generated {STRUCTURE_TXT}")

    # 2. Aggregate
    paths = read_file_paths(files_txt)
    if not paths:
        print(f"No files listed in {FILES_TXT}. Nothing to do.")
        return

    count = aggregate_files(paths, root, arena_txt)
    print(f"Aggregated {count} files into {ARENA_TXT}")


if __name__ == "__main__":
    main()
```

```python
"""Terminal User Interface for File Aggregator (Command: aggt)."""

from pathlib import Path
from typing import List, Set
from textual.app import App, ComposeResult
from textual.containers import Container, Horizontal, Vertical
from textual.widgets import Header, Footer, Button, Label, Tree, Static, Log
from textual.widgets.tree import TreeNode
import aggregator as core

class FileTree(Tree[str]):
    """Custom Tree that populates lazily."""
    
    def __init__(self, root: Path, **kwargs):
        super().__init__(str(root), data=str(root), **kwargs)
        self.root_path = root
        self.patterns = core.load_ignore_patterns(root)

    def on_mount(self) -> None:
        self._populate_node(self.root, self.root_path)

    def _populate_node(self, node: TreeNode, path: Path) -> None:
        node.remove_children()
        try:
            entries = sorted(path.iterdir(), key=lambda p: (p.is_file(), p.name.lower()))
        except PermissionError:
            return

        for entry in entries:
            if core.should_ignore(entry, self.root_path, self.patterns):
                continue
            
            label = entry.name
            if entry.is_dir():
                # Add directory node with placeholder
                child = node.add(label, data=str(entry))
                child.add("Loading...", data="placeholder")
            else:
                node.add_leaf(label, data=str(entry))

    def on_tree_node_expanded(self, event: Tree.NodeExpanded) -> None:
        node = event.node
        if node.data == "placeholder":
            return
        path = Path(node.data)
        if path.is_dir():
            self._populate_node(node, path)


class AggregatorTUI(App):
    CSS = """

    #tree-col { border-right: solid $primary; overflow: auto; }

    """

    BINDINGS = [
        ("a", "aggregate", "Aggregate"),
        ("q", "quit", "Quit")
    ]

    def __init__(self):
        super().__init__()
        self.root = core.find_project_root()
        self.selected: Set[Path] = set()

    def compose(self) -> ComposeResult:
        yield Header()
        yield Container(
            Vertical(
                Label(f"Project: {self.root.name}"),
                FileTree(self.root, id="tree"),
                id="tree-col"
            ),
            Vertical(
                Label("Activity Log"),
                Log(id="log"),
                Horizontal(
                    Button("Aggregate", variant="primary", id="btn_agg"),
                    Button("Clear Selection", id="btn_clear"),
                    id="controls"
                ),
                id="info-col"
            )
        )
        yield Footer()

    def on_tree_node_selected(self, event: Tree.NodeSelected) -> None:
        path = Path(event.node.data)
        if path.is_file():
            if path in self.selected:
                self.selected.remove(path)
                self._log(f"Deselected: {path.name}")
            else:
                self.selected.add(path)
                self._log(f"Selected: {path.name}")
            # Update node label style (simple approach: just visual feedback in log)
            # Textual Tree styling requires CSS classes, simplified here.

    def on_button_pressed(self, event: Button.Pressed) -> None:
        if event.button.id == "btn_agg":
            self.action_aggregate()
        elif event.button.id == "btn_clear":
            self.selected.clear()
            self._log("Selection cleared.")

            self._log("Error: No files selected.")
            return
        
        self._log(f"Aggregating {len(self.selected)} files...")
        arena_path = self.root / core.ARENA_TXT
        
        # Run aggregation
        count = core.aggregate_files(list(self.selected), self.root, arena_path)
        self._log(f"Success: Aggregated {count} files -> {core.ARENA_TXT}")

        log = self.query_one("#log", Log)
        log.write_line(msg)

if __name__ == "__main__":
```

```python
        main_pane.add(left_frame, weight=3)

        # Right: Log & Controls
        right_frame = ttk.Frame(main_pane)
        self.log_text = tk.Text(right_frame, height=20, state="disabled", bg="#1e1e1e", fg="#d4d4d4", font=("Consolas", 10))
        self.log_text.pack(fill="both", expand=True, pady=(0, 10))

        btn_frame = ttk.Frame(right_frame)
        btn_frame.pack(fill="x")
        
        ttk.Button(btn_frame, text="Aggregate Selected", command=self._start_aggregation).pack(side="left", padx=5)
        ttk.Button(btn_frame, text="Clear Selection", command=self._clear_selection).pack(side="left")

        main_pane.add(right_frame, weight=2)

        self.root.protocol("WM_DELETE_WINDOW", self.root.destroy)

    def _populate_tree(self, parent="", path=None):
        """Recursively populate tree view."""
        path = path or self.project_root
        node_id = self.tree.insert(parent, "end", text=path.name, open=False)
        
        try:
            entries = sorted(path.iterdir(), key=lambda p: (p.is_file(), p.name.lower()))
        except PermissionError:
            return

        for entry in entries:
            if core.should_ignore(entry, self.project_root, self.ignore_patterns):
                continue
            
            if entry.is_dir():
                # Recursive call to populate children
                self._populate_tree(node_id, entry)
            else:
                self.tree.insert(node_id, "end", text=entry.name, values=(str(entry),))

        item_text = self.tree.item(item_id, "text")

        full_path = self.project_root / item_text # Simplified logic

        if not full_path.is_dir():
            if full_path in self.selected_files:
                self.selected_files.remove(full_path)
                self._log(f"Removed: {item_text}")
            else:
                self.selected_files.add(full_path)
                self._log(f"Added: {item_text}")

        self.log_text.insert("end", f"> {msg}\n")
        self.log_text.see("end")
        self.log_text.config(state="disabled")

    def run(self):
        self.root.mainloop()

if __name__ == "__main__":
    app = AggregatorGUI()
    app.run()
```

```python
    a   Run aggregation in background thread.
    c   Clear queue and uncheck all entries.
    q   Quit.
    """

    TITLE = "File Aggregator"
    CSS = _CSS

    BINDINGS: ClassVar[list[Binding]] = [
        Binding("r", "refresh", "Refresh", show=True),
        Binding("a", "aggregate", "Aggregate", show=True),
        Binding("c", "clear", "Clear Queue", show=True),
        Binding("q", "quit", "Quit", show=True),
    ]

    _queue_count: reactive[int] = reactive(0)

    # ------------------------------------------------------------------
    # Lifecycle
    # ------------------------------------------------------------------

    def on_mount(self) -> None:
        """Populate tree and queue on startup."""
        self._load_tree()
        self._load_queue()

    # ------------------------------------------------------------------
    # Layout
    # ------------------------------------------------------------------

    def compose(self) -> ComposeResult:
        yield Header()

        with Horizontal(id="body"):
            with Vertical(id="tree-panel"):
                yield Static("📁  Project Tree  (click to queue)", id="tree-title")
                with ScrollableContainer(id="tree-scroll"):
                    pass

            with Vertical(id="right-panel"):
                with Vertical(id="queue-panel"):
                    yield Static("📋  Queue  (0 files)", id="queue-title")
                    with ScrollableContainer(id="queue-scroll"):
                        pass

                with Vertical(id="log-panel"):
                    yield Log(id="activity-log", auto_scroll=True)

        with Horizontal(id="controls"):
            yield Button("⟳  Refresh", id="btn-refresh", variant="default")
            yield Button("▶  Aggregate", id="btn-aggregate", variant="primary")
            yield Button("✕  Clear", id="btn-clear", variant="warning")
            yield Button("⏻  Quit", id="btn-quit", variant="error")

        yield Footer()

    # ------------------------------------------------------------------
    # Tree population
    # ------------------------------------------------------------------

    def _resolve_root(self) -> Path:
        """Return project root inferred from files.txt, or fall back to _PROJECT_DIR."""
        if _FILES_TXT.is_file():
            try:
                paths = read_file_paths(_FILES_TXT)
                if paths:
                    candidate = find_project_root(paths[0])
                    if candidate:
                        return candidate
            except Exception:  # noqa: BLE001
                pass
        return _PROJECT_DIR

    def _load_tree(self) -> None:
        """Scan the project directory and rebuild the tree panel."""
        log = self.query_one("#activity-log", Log)
        scroll = self.query_one("#tree-scroll", ScrollableContainer)
        scroll.remove_children()

        root = self._resolve_root()
        patterns = load_ignore_patterns(root)
        self._populate_tree(scroll, root, root, patterns)
        log.write_line(f"[tree] root: {root}")

    def _populate_tree(
        self,
        container: ScrollableContainer,
        dir_path: Path,
        root: Path,
        patterns: frozenset[str],
        depth: int = 0,
    ) -> None:
        """Recursively mount TreeEntry / Label widgets for *dir_path*.

        Args:
            container: Receives new child widgets.
            dir_path:  Directory currently being scanned.
            root:      Project root used for ignore-pattern evaluation.
            patterns:  Active ignore patterns.
            depth:     Current indentation depth (hard limit: 6).
        """
        if depth > 6:
            return

        try:
            items = sorted(
                dir_path.iterdir(),
                key=lambda p: (not p.is_dir(), p.name.lower()),
            )
        except PermissionError:
            return

        for item in items:
            if should_ignore(item, root, patterns):
                continue

            indent = "  " * depth

            if item.is_dir() and not item.is_symlink():
                container.mount(Label(f"{indent}📁 {item.name}/", classes="tree-dir"))
                self._populate_tree(container, item, root, patterns, depth + 1)

            elif item.is_file():
                entry = TreeEntry(
                    f"{indent}  {item.name}",
                    file_path=item.resolve(),
                    value=self._path_in_queue(item.resolve()),
                )
                container.mount(entry)

    def _path_in_queue(self, path: Path) -> bool:
        """Return True when *path* is already recorded in files.txt."""
        if not _FILES_TXT.is_file():
            return False
        try:
            return path in {p.resolve() for p in read_file_paths(_FILES_TXT)}
        except Exception:  # noqa: BLE001
            return False

    # ------------------------------------------------------------------
    # Queue display
    # ------------------------------------------------------------------

    def _load_queue(self) -> None:
        """Rebuild the queue panel from files.txt."""
        scroll = self.query_one("#queue-scroll", ScrollableContainer)
        title = self.query_one("#queue-title", Static)
        log = self.query_one("#activity-log", Log)

        scroll.remove_children()

        if not _FILES_TXT.is_file():
            self._queue_count = 0
            title.update("📋  Queue  (0 files)")
            return

        try:
            paths = read_file_paths(_FILES_TXT)
        except Exception as exc:  # noqa: BLE001
            log.write_line(f"[error] reading files.txt: {exc}")
            return

        for p in paths:
            scroll.mount(Label(str(p)))

        self._queue_count = len(paths)
        title.update(f"📋  Queue  ({self._queue_count} files)")
        log.write_line(f"[queue] {self._queue_count} file(s).")

    # ------------------------------------------------------------------
    # Event handlers
    # ------------------------------------------------------------------

    @on(Checkbox.Changed)
    def handle_checkbox(self, event: Checkbox.Changed) -> None:
        """Persist a checkbox toggle to files.txt.

        Args:
            event: Carries the changed Checkbox widget and its new value.
        """
        if not isinstance(event.checkbox, TreeEntry):
            return

        entry: TreeEntry = event.checkbox

        # Skip write when action_clear() is driving the toggle.
        if entry._suppress_write:
            return

        log = self.query_one("#activity-log", Log)
        self._update_files_txt(entry.file_path, add=event.value)
        verb = "Added" if event.value else "Removed"
        log.write_line(f"[queue] {verb}: {entry.file_path.name}")
        self._load_queue()

    @on(Button.Pressed, "#btn-refresh")
    def _btn_refresh(self) -> None:
        self.action_refresh()

    @on(Button.Pressed, "#btn-aggregate")
    def _btn_aggregate(self) -> None:
        self.action_aggregate()

    @on(Button.Pressed, "#btn-clear")
    def _btn_clear(self) -> None:
        self.action_clear()

    @on(Button.Pressed, "#btn-quit")
    def _btn_quit(self) -> None:
        self.action_quit()

    # ------------------------------------------------------------------
    # Actions
    # ------------------------------------------------------------------

    def action_refresh(self) -> None:
        """Re-scan the filesystem and reload the queue display."""
        self._load_tree()
        self._load_queue()

    @work(thread=True)
    def action_aggregate(self) -> None:
        """Run aggregation in a background worker thread (non-blocking).

        Writes structure.txt and arena.txt; posts results to the activity log.
        """
        log = self.query_one("#activity-log", Log)
        self.call_from_thread(log.write_line, "[run] Starting aggregation…")

        try:
            paths = read_file_paths(_FILES_TXT)
        except FileNotFoundError:
            self.call_from_thread(
                log.write_line, "[error] files.txt not found — add files first."
            )
            return

        if not paths:
            self.call_from_thread(log.write_line, "[warn] files.txt is empty.")
            return

        try:
            root = find_project_root(paths[0])
            patterns = load_ignore_patterns(root)

            if root:
                tree_lines = [f"Project Root: {root.name}/"] + generate_tree(
                    root, root, patterns
                )
                _STRUCTURE_TXT.write_text("\n".join(tree_lines), encoding="utf-8")
                self.call_from_thread(
                    log.write_line, f"[ok] structure.txt written ({root.name})"
                )

            aggregate_files(paths, _ARENA_TXT, root)
            self.call_from_thread(
                log.write_line,
                f"[ok] arena.txt written ({len(paths)} file(s)).",
            )
        except Exception as exc:  # noqa: BLE001
            self.call_from_thread(log.write_line, f"[error] {exc}")

    def action_clear(self) -> None:
        """Empty files.txt and uncheck every tree entry without triggering file writes."""
        _FILES_TXT.write_text("", encoding="utf-8")

        for entry in self.query(TreeEntry):
            entry._suppress_write = True
            entry.value = False
            entry._suppress_write = False

        log = self.query_one("#activity-log", Log)
        log.write_line("[queue] Cleared.")
        self._load_queue()

    # ------------------------------------------------------------------
    # files.txt mutation
    # ------------------------------------------------------------------

    def _update_files_txt(self, path: Path, *, add: bool) -> None:
        """Add or remove *path* from files.txt atomically.

        Args:
            path: The resolved file path to add or remove.
            add:  True to add, False to remove.
        """
        existing: list[Path] = []
        if _FILES_TXT.is_file():
            try:
                existing = [p.resolve() for p in read_file_paths(_FILES_TXT)]
            except Exception:  # noqa: BLE001
                pass

        if add:
            if path not in existing:
                existing.append(path)
        else:
            existing = [p for p in existing if p != path]

        _FILES_TXT.write_text(
            "\n".join(str(p) for p in existing) + ("\n" if existing else ""),
            encoding="utf-8",
        )


def main() -> None:
    """Launch the TUI."""
    AggregatorTUI().run()


if __name__ == "__main__":
    main()
```

```python
    def _insert_children(self, parent_iid: str, dir_path: Path) -> None:
        """Populate *parent_iid* with the direct children of *dir_path*.

        Directories receive a placeholder child so Tkinter renders the
        expand arrow; real children are injected on ``<<TreeviewOpen>>``.

        Args:
            parent_iid: Treeview node that will receive new children.
            dir_path:   Filesystem directory to enumerate.
        """
        try:
            items = sorted(
                dir_path.iterdir(),
                key=lambda p: (not p.is_dir(), p.name.lower()),
            )
        except PermissionError:
            return

        for item in items:
            if should_ignore(item, self._root_path, self._patterns):
                continue

            if item.is_dir() and not item.is_symlink():
                iid = self._tree.insert(
                    parent_iid, "end", text=f"📁 {item.name}/"
                )
                # Placeholder keeps the expand arrow visible
                self._tree.insert(iid, "end", text="…")

            elif item.is_file():
                resolved = item.resolve()
                marker = "✔ " if resolved in self._queued_paths else "   "
                iid = self._tree.insert(
                    parent_iid,
                    "end",
                    text=f"{marker}{item.name}",
                )
                self._iid_to_path[iid] = resolved

    def _on_tree_open(self, _event: object) -> None:
        """Lazily expand a directory node on first open.

        Removes the placeholder child and inserts real filesystem entries.
        """
        iid = self._tree.focus()
        children = self._tree.get_children(iid)

        # Detect placeholder: single child with text "…"
        if len(children) == 1 and self._tree.item(children[0], "text") == "…":
            self._tree.delete(children[0])
            # Recover directory path from the parent's own children mapping.
            # We reconstruct from the label text walking up to root.
            dir_path = self._iid_label_to_path(iid)
            if dir_path and dir_path.is_dir():
                self._insert_children(iid, dir_path)

    def _iid_label_to_path(self, iid: str) -> Path | None:
        """Reconstruct a directory Path by walking ancestor labels up to root.

        This avoids storing directory paths in _iid_to_path (which is
        reserved for files only, so queue-lookup stays O(1)).

        Args:
            iid: Treeview node identifier for a directory node.

        Returns:
            The resolved directory Path, or None if reconstruction fails.
        """
        parts: list[str] = []
        current = iid
        while current:
            raw = self._tree.item(current, "text")
            # Strip decoration: "📁 name/" → "name"
            name = raw.lstrip("📁 ").rstrip("/").strip()
            parts.append(name)
            current = self._tree.parent(current)

        parts.reverse()
        # parts[0] is the root node label (== self._root_path.name)
        candidate = self._root_path.parent
        for part in parts:
            candidate = candidate / part

        return candidate.resolve() if candidate.exists() else None

    # ------------------------------------------------------------------
    # Queue management
    # ------------------------------------------------------------------

    def _current_queue_paths(self) -> set[Path]:
        """Return the set of resolved paths in files.txt."""
        if not _FILES_TXT.is_file():
            return set()
        try:
            return {p.resolve() for p in read_file_paths(_FILES_TXT)}
        except Exception:  # noqa: BLE001
            return set()

    def _refresh_queue(self) -> None:
        """Reload the queue listbox from files.txt."""
        self._queue_listbox.delete(0, "end")
        paths = sorted(self._current_queue_paths(), key=lambda p: p.name)
        for path in paths:
            self._queue_listbox.insert("end", str(path))
        self._queue_title.configure(text=f"📋  Queue  ({len(paths)} files)")

    def _write_queue(self, paths: set[Path]) -> None:
        """Overwrite files.txt with *paths* in sorted order.

        Args:
            paths: Complete desired queue contents.
        """
        ordered = sorted(paths, key=str)
        _FILES_TXT.write_text(
            "\n".join(str(p) for p in ordered) + ("\n" if ordered else ""),
            encoding="utf-8",
        )

    def _add_selected(self) -> None:
        """Add every tree-selected file to the queue."""
        selection = self._tree.selection()
        if not selection:
            self._log_write("No items selected in the tree.", tag="warn")
            return

        current = self._current_queue_paths()
        added = 0
        for iid in selection:
            path = self._iid_to_path.get(iid)
            if path and path.is_file() and path not in current:
                current.add(path)
                added += 1

        self._write_queue(current)
        self._log_write(f"Added {added} file(s).", tag="ok")
        self._refresh_queue()
        self._refresh_tree()

    def _remove_selected(self) -> None:
        """Remove highlighted listbox entries from files.txt."""
        indices = self._queue_listbox.curselection()
        if not indices:
            self._log_write("No items selected in the queue.", tag="warn")
            return

        current = self._current_queue_paths()
        to_remove = {Path(self._queue_listbox.get(i)) for i in indices}
        current -= to_remove
        self._write_queue(current)
        self._log_write(f"Removed {len(to_remove)} file(s).", tag="warn")
        self._refresh_queue()
        self._refresh_tree()

    def _clear_queue(self) -> None:
        """Empty files.txt after confirmation."""
        if not messagebox.askyesno(
            "Confirm Clear",
            "Remove all files from the queue?\nThis cannot be undone.",
        ):
            return
        _FILES_TXT.write_text("", encoding="utf-8")
        self._log_write("Queue cleared.", tag="warn")
        self._refresh_queue()
        self._refresh_tree()

    def _open_root(self) -> None:
        """Let the user pick a new project root directory."""
        chosen = filedialog.askdirectory(title="Select Project Root")
        if chosen:
            _FILES_TXT.write_text("", encoding="utf-8")
            self._log_write(f"Root changed to: {chosen}", tag="info")
            self._refresh_tree()
            self._refresh_queue()

    # ------------------------------------------------------------------
    # Aggregation (daemon thread)
    # ------------------------------------------------------------------

    def _run_aggregate(self) -> None:
        """Dispatch aggregation to a daemon thread."""
        threading.Thread(target=self._aggregate_worker, daemon=True).start()

    def _aggregate_worker(self) -> None:
        """Aggregation body — safe to call off the main thread."""
        self._set_status("Aggregating…")
        self._log_write("Starting aggregation…", tag="info")

        try:
            paths = read_file_paths(_FILES_TXT)
        except FileNotFoundError:
            self._log_write("files.txt not found — add files first.", tag="error")
            self._set_status("Error.")
            return

        if not paths:
            self._log_write("files.txt is empty — nothing to aggregate.", tag="warn")
            self._set_status("Nothing to do.")
            return

        try:
            root = find_project_root(paths[0])
            patterns = load_ignore_patterns(root)

            if root:
                tree_lines = [f"Project Root: {root.name}/"] + generate_tree(
                    root, root, patterns
                )
                _STRUCTURE_TXT.write_text("\n".join(tree_lines), encoding="utf-8")
                self._log_write(f"structure.txt written  ({root.name})", tag="ok")

            aggregate_files(paths, _ARENA_TXT, root)
            self._log_write(
                f"arena.txt written  ({len(paths)} file(s)).", tag="ok"
            )
            self._set_status(f"Done — {len(paths)} file(s) aggregated.")

        except Exception as exc:  # noqa: BLE001
            self._log_write(f"Error: {exc}", tag="error")
            self._set_status("Error.")

    # ------------------------------------------------------------------
    # Thread-safe UI helpers
    # ------------------------------------------------------------------

    def _log_write(self, message: str, tag: str = "") -> None:
        """Append *message* to the activity log from any thread.

        Args:
            message: Text to display.
            tag:     One of ``"ok"``, ``"warn"``, ``"error"``, ``"info"``.
        """
        def _insert() -> None:
            self._log.configure(state="normal")
            self._log.insert("end", f"› {message}\n", tag or "")
            self._log.see("end")
            self._log.configure(state="disabled")

        self.after(0, _insert)

    def _set_status(self, text: str) -> None:
        """Update the status-bar label from any thread.

        Args:
            text: Status string to display.
        """
        self.after(0, lambda: self._status_var.set(text))


def main() -> None:
    """Launch the GUI."""
    AggregatorGUI().run()  # type: ignore[attr-defined]


if __name__ == "__main__":
    AggregatorGUI().mainloop()
```

| Issue | Source | Fix applied |
|---|---|---|
| `_suppress_write` guard missing | All three | `TreeEntry._suppress_write` flag prevents `action_clear()` from triggering N file writes |
| `yscroll=` TclError | gpt-5.2 | Corrected to `yscrollcommand=` |
| Tree paths reconstructed from ASCII art | grok | Replaced with `_iid_to_path` dict (files) + `_iid_label_to_path()` walk (directories) |
| Directory expand blocks event loop | sonnet | `_on_tree_open` + placeholder pattern makes expansion lazy and non-blocking |
| Path resolve inconsistency | sonnet | All paths normalized with `.resolve()` before set membership tests |
| `main()` calls `.run()` on Tk subclass | sonnet | Entry point calls `.mainloop()` directly; `main()` kept for alias compatibility |
| Speculative `aggregator.main()` import | gpt-5.2 | Removed; both UIs call the imported functions directly since the API is known |"
gpt-5.2-high : "Below are **drop-in** implementations of the two UI files from the “Dual‑Interface Enhancement” plan, with two practical fixes applied:

1) **TUI clear operation won’t spam `files.txt` writes** (suppresses checkbox events during programmatic uncheck).  
2) **GUI “Open Root” actually changes the root** (keeps a `_root` state instead of immediately losing the chosen directory), and “Refresh” refreshes **both** tree + queue.

- `should_ignore(path: Path, root: Path, patterns: frozenset[str]) -> bool`

```python
class AggregatorTUI(App[None]):
    """Interactive TUI for browsing, selecting, and aggregating project files.

    Keyboard shortcuts:
        r   Refresh the project tree.
        a   Run aggregation.

    def __init__(self) -> None:
        super().__init__()
        # Prevent programmatic checkbox changes (e.g., "Clear") from rewriting files.txt
        self._suppress_checkbox_events: bool = False

    def on_mount(self) -> None:

        self._load_queue()

    def compose(self) -> ComposeResult:

            yield Button("✕  Clear", id="btn-clear", variant="warning")
            yield Button("⏻  Quit", id="btn-quit", variant="error")

        yield Footer()

    # ------------------------------------------------------------------
    # Tree loading
    # ------------------------------------------------------------------

    def _detect_root(self) -> Path:
        """Detect root from first queued path if possible, otherwise use project dir."""
        if _FILES_TXT.is_file():
            try:
                paths = read_file_paths(_FILES_TXT)
                if paths:
                    root = find_project_root(paths[0])
                    if root:
                        return root
            except Exception:  # noqa: BLE001
                pass
        return _PROJECT_DIR

    def _load_tree(self) -> None:
        log = self.query_one("#activity-log", Log)
        scroll = self.query_one("#tree-scroll", ScrollableContainer)
        scroll.remove_children()

        root = self._detect_root()
        patterns = load_ignore_patterns(root)

        self._populate_tree(scroll, root, root, patterns)
        log.write_line(f"[tree] Loaded from: {root}")

    def _populate_tree(
        self,
        container: ScrollableContainer,
        dir_path: Path,
        root: Path,
        patterns: frozenset[str],
        depth: int = 0,
    ) -> None:
        if depth > 6:
            return

        try:
            items = sorted(
                dir_path.iterdir(),
                key=lambda p: (not p.is_dir(), p.name.lower()),
            )
        except PermissionError:
            return

        queued = set()
        if _FILES_TXT.is_file():
            try:
                queued = {p.resolve() for p in read_file_paths(_FILES_TXT)}
            except Exception:  # noqa: BLE001
                queued = set()

        for item in items:
            if should_ignore(item, root, patterns):
                continue

            indent = "  " * depth
            if item.is_dir() and not item.is_symlink():
                container.mount(Label(f"{indent}📁 {item.name}/"))
                self._populate_tree(container, item, root, patterns, depth + 1)
            elif item.is_file():
                entry = TreeEntry(
                    f"{indent}  {item.name}",
                    file_path=item,
                    value=item.resolve() in queued,
                )
                container.mount(entry)

    # ------------------------------------------------------------------
    # Queue loading
    # ------------------------------------------------------------------

    def _load_queue(self) -> None:
        scroll = self.query_one("#queue-scroll", ScrollableContainer)
        title = self.query_one("#queue-title", Static)
        log = self.query_one("#activity-log", Log)

        scroll.remove_children()

        if not _FILES_TXT.is_file():
            self._queue_count = 0
            title.update("📋  Queue  (0 files)")
            return

        try:
            paths = read_file_paths(_FILES_TXT)
        except Exception as exc:  # noqa: BLE001
            log.write_line(f"[error] Could not read files.txt: {exc}")
            return

        for path in paths:
            scroll.mount(Label(str(path)))

        self._queue_count = len(paths)
        title.update(f"📋  Queue  ({self._queue_count} files)")
        log.write_line(f"[queue] {self._queue_count} file(s) loaded.")

    # ------------------------------------------------------------------
    # Events
    # ------------------------------------------------------------------

    @on(Checkbox.Changed)
    def handle_checkbox(self, event: Checkbox.Changed) -> None:
        if self._suppress_checkbox_events:
            return
        if not isinstance(event.checkbox, TreeEntry):
            return

        entry = event.checkbox
        log = self.query_one("#activity-log", Log)

        self._update_files_txt(entry.file_path, add=event.value)
        log.write_line(f"[queue] {'Added' if event.value else 'Removed'}: {entry.file_path.name}")
        self._load_queue()

    @on(Button.Pressed, "#btn-refresh")
    def handle_refresh(self) -> None:
        self.action_refresh()

    @on(Button.Pressed, "#btn-aggregate")
    def handle_aggregate(self) -> None:
        self.action_aggregate()

    @on(Button.Pressed, "#btn-clear")
    def handle_clear(self) -> None:
        self.action_clear()

    @on(Button.Pressed, "#btn-quit")
    def handle_quit(self) -> None:
        self.action_quit()

    # ------------------------------------------------------------------
    # Actions
    # ------------------------------------------------------------------

    def action_refresh(self) -> None:
        self._load_tree()
        self._load_queue()

    @work(thread=True)
    def action_aggregate(self) -> None:
        log = self.query_one("#activity-log", Log)
        self.call_from_thread(log.write_line, "[run] Starting aggregation…")

        try:
            paths = read_file_paths(_FILES_TXT)
            if not paths:
                self.call_from_thread(log.write_line, "[warn] files.txt is empty — nothing to do.")
                return

            root = find_project_root(paths[0])
            patterns = load_ignore_patterns(root or self._detect_root())

            if root:
                tree_lines = [f"Project Root: {root.name}/"] + generate_tree(root, root, patterns)
                _STRUCTURE_TXT.write_text("\n".join(tree_lines), encoding="utf-8")
                self.call_from_thread(log.write_line, "[ok] structure.txt written")

            aggregate_files(paths, _ARENA_TXT, root)
            self.call_from_thread(log.write_line, f"[ok] arena.txt written ({len(paths)} file(s)).")

        except FileNotFoundError:
            self.call_from_thread(log.write_line, "[error] files.txt not found — add files first.")
        except Exception as exc:  # noqa: BLE001
            self.call_from_thread(log.write_line, f"[error] {exc}")

    def action_clear(self) -> None:
        _FILES_TXT.write_text("", encoding="utf-8")

        self._suppress_checkbox_events = True
        try:
            for entry in self.query(TreeEntry):
                entry.value = False
        finally:
            self._suppress_checkbox_events = False

        log = self.query_one("#activity-log", Log)
        log.write_line("[queue] Cleared.")
        self._load_queue()

    # ------------------------------------------------------------------
    # files.txt mutation
    # ------------------------------------------------------------------

    def _update_files_txt(self, path: Path, *, add: bool) -> None:
        existing: list[Path] = []
        if _FILES_TXT.is_file():
            try:
                existing = read_file_paths(_FILES_TXT)
            except Exception:  # noqa: BLE001
                existing = []

        # Normalize for stable comparisons
        path = path.resolve()
        existing_norm = [p.resolve() for p in existing]

        if add:
            if path not in existing_norm:
                existing.append(path)
        else:
            existing = [p for p in existing if p.resolve() != path]

        _FILES_TXT.write_text(
            "\n".join(str(p) for p in existing) + ("\n" if existing else ""),
            encoding="utf-8",
        )

    """Launch the TUI application."""
    AggregatorTUI().run()

    main()
```

```python
        queued = self._current_queue_paths()

        self._tree_title.configure(text=f"📁  Project Tree  ({root})")
        self._populate_tree("", root, root, patterns, queued)
        self._log_write(f"Tree loaded from: {root}", tag="info")

    def _populate_tree(
        self,
        parent_iid: str,
        dir_path: Path,
        root: Path,
        patterns: frozenset[str],
        queued: set[Path],
        depth: int = 0,
    ) -> None:
        if depth > 8:
            return

        try:
            items = sorted(
                dir_path.iterdir(),
                key=lambda p: (not p.is_dir(), p.name.lower()),
            )
        except PermissionError:
            return

        queued_resolved = {p.resolve() for p in queued}

        for item in items:
            if should_ignore(item, root, patterns):
                continue

            if item.is_dir() and not item.is_symlink():
                iid = self._tree.insert(
                    parent_iid,
                    "end",
                    text=f"📁 {item.name}",
                    open=False,
                    values=[str(item)],
                )
                self._populate_tree(iid, item, root, patterns, queued, depth + 1)
            elif item.is_file():
                marker = "✔ " if item.resolve() in queued_resolved else "  "
                self._tree.insert(
                    parent_iid,
                    "end",
                    text=f"{marker}{item.name}",
                    values=[str(item)],
                )

    # ------------------------------------------------------------------
    # Queue management
    # ------------------------------------------------------------------

    def _current_queue_paths(self) -> set[Path]:
        if not _FILES_TXT.is_file():
            return set()
        try:
            return set(read_file_paths(_FILES_TXT))
        except Exception:  # noqa: BLE001
            return set()

    def _refresh_queue(self) -> None:
        self._queue_listbox.delete(0, "end")
        paths = sorted(self._current_queue_paths(), key=lambda p: str(p))
        for path in paths:
            self._queue_listbox.insert("end", str(path))
        self._queue_title.configure(text=f"📋  Queue  ({len(paths)} files)")

    def _write_queue(self, paths: set[Path]) -> None:
        ordered = sorted({p.resolve() for p in paths}, key=lambda p: str(p))
        _FILES_TXT.write_text(
            "\n".join(str(p) for p in ordered) + ("\n" if ordered else ""),
            encoding="utf-8",
        )

    def _add_selected(self) -> None:
        selection = self._tree.selection()
        if not selection:
            self._log_write("No items selected in the tree.", tag="warn")
            return

        current = self._current_queue_paths()
        added = 0
        for iid in selection:
            values = self._tree.item(iid, "values")
            if not values:
                continue
            path = Path(values[0])
            if path.is_file() and path.resolve() not in {p.resolve() for p in current}:
                current.add(path.resolve())
                added += 1

        self._write_queue(current)
        self._log_write(f"Added {added} file(s) to queue.", tag="ok")
        self._refresh_all()

    def _remove_selected(self) -> None:
        indices = self._queue_listbox.curselection()
        if not indices:
            self._log_write("No items selected in the queue.", tag="warn")
            return

        current = {p.resolve() for p in self._current_queue_paths()}
        to_remove = {Path(self._queue_listbox.get(i)).resolve() for i in indices}
        current -= to_remove

        self._write_queue(current)
        self._log_write(f"Removed {len(to_remove)} file(s).", tag="warn")
        self._refresh_all()

    def _clear_queue(self) -> None:
        if not messagebox.askyesno(
            "Confirm Clear",
            "Remove all files from the queue?\nThis cannot be undone.",
        ):
            return
        _FILES_TXT.write_text("", encoding="utf-8")
        self._log_write("Queue cleared.", tag="warn")
        self._refresh_all()

    def _open_root(self) -> None:
        chosen = filedialog.askdirectory(title="Select Project Root")
        if not chosen:
            return
        self._root = Path(chosen).resolve()
        _FILES_TXT.write_text("", encoding="utf-8")  # reset queue
        self._log_write(f"Root changed to: {self._root}", tag="info")
        self._refresh_all()

    # ------------------------------------------------------------------
    # Aggregation (background thread)
    # ------------------------------------------------------------------

    def _run_aggregate(self) -> None:
        threading.Thread(target=self._aggregate_worker, daemon=True).start()

    def _aggregate_worker(self) -> None:
        self._set_status("Aggregating…")
        self._log_write("Starting aggregation…", tag="info")

        try:
            paths = read_file_paths(_FILES_TXT)
            if not paths:
                self._log_write("files.txt is empty — nothing to aggregate.", tag="warn")
                self._set_status("Nothing to do.")
                return

            root = find_project_root(paths[0]) or self._root
            patterns = load_ignore_patterns(root)

            tree_lines = [f"Project Root: {root.name}/"] + generate_tree(root, root, patterns)
            _STRUCTURE_TXT.write_text("\n".join(tree_lines), encoding="utf-8")
            self._log_write(f"structure.txt written ({root.name})", tag="ok")

            aggregate_files(paths, _ARENA_TXT, root)
            self._log_write(f"arena.txt written ({len(paths)} file(s)).", tag="ok")
            self._set_status(f"Done — {len(paths)} file(s) aggregated.")

        except FileNotFoundError:
            self._log_write("files.txt not found — add files first.", tag="error")
            self._set_status("Error.")
        except Exception as exc:  # noqa: BLE001
            self._log_write(f"Error: {exc}", tag="error")
            self._set_status("Error.")

    # ------------------------------------------------------------------
    # Thread-safe helpers
    # ------------------------------------------------------------------

    def _log_write(self, message: str, tag: str = "") -> None:
        def _insert() -> None:
            self._log.configure(state="normal")
            self._log.insert("end", f"› {message}\n", tag or "")
            self._log.see("end")
            self._log.configure(state="disabled")

        self.after(0, _insert)

    def _set_status(self, text: str) -> None:
        self.after(0, lambda: self._status_var.set(text))
```

```python
"""Pure library for the File Aggregator tool.

All three interfaces (agg, aggt, aggg) import from this module.
No side effects on import.
"""

import fnmatch
import sys
from pathlib import Path
from typing import List, Set, FrozenSet

_ROOT_MARKERS: FrozenSet[str] = frozenset({
    ".git", "package.json", "pyproject.toml", "requirements.txt", "src"
})

_DEFAULT_IGNORE: FrozenSet[str] = frozenset({
    ".git", "node_modules", "dist", "__pycache__", ".next", "build",
    ".venv", ".env", "*.pyc", ".DS_Store", ".index_ignore"
})

_MAX_TREE_DEPTH = 8


def find_project_root(start: Path) -> Path:
    """Return nearest project root or fall back to start parent."""
    current = start.resolve()
    while current != current.parent:
        if any((current / marker).exists() for marker in _ROOT_MARKERS):
            return current
        current = current.parent
    return start.resolve().parent


def load_ignore_patterns(root: Path) -> FrozenSet[str]:
    """Load .index_ignore (if present) and merge with defaults."""
    patterns: Set[str] = set(_DEFAULT_IGNORE)
    ignore_file = root / ".index_ignore"

    if ignore_file.is_file():
        try:
            for line in ignore_file.read_text(encoding="utf-8").splitlines():
                stripped = line.strip()
                if stripped and not stripped.startswith("#"):
                    patterns.add(stripped)
        except OSError as exc:
            print(f"Warning: Could not read .index_ignore: {exc}", file=sys.stderr)

    return frozenset(patterns)


def should_ignore(path: Path, root: Path, patterns: FrozenSet[str]) -> bool:
    """Test whether a path should be ignored."""
    try:
        rel = path.relative_to(root).as_posix()
        name = path.name
    except ValueError:
        return False

    return any(
        fnmatch.fnmatch(rel, pat) or fnmatch.fnmatch(name, pat)
        for pat in patterns
    )


def generate_tree(
    dir_path: Path, root: Path, patterns: FrozenSet[str], depth: int = 0
) -> List[str]:
    """Generate a visual tree for structure.txt."""
    if depth > _MAX_TREE_DEPTH:
        return ["    " * depth + "... (max depth reached)"]

    tree: List[str] = []
    try:
        items = sorted(
            (p for p in dir_path.iterdir() if not should_ignore(p, root, patterns)),
            key=lambda p: (not p.is_dir(), p.name.lower()),
        )
    except (PermissionError, OSError):
        return tree

    for i, item in enumerate(items):
        is_last = i == len(items) - 1
        prefix = "└── " if is_last else "├── "
        tree.append("    " * depth + prefix + item.name + ("/" if item.is_dir() else ""))

        if item.is_dir() and not item.is_symlink():
            new_prefix = "    " * depth + ("    " if is_last else "│   ")
            tree.extend(generate_tree(item, root, patterns, depth + 1))

    return tree


def read_file_paths(source: Path) -> List[Path]:
    """Read absolute paths from files.txt."""
    if not source.is_file():
        return []
    try:
        return [Path(line.strip()) for line in source.read_text(encoding="utf-8").splitlines() if line.strip()]
    except OSError:
        return []


def aggregate_files(paths: List[Path], output_file: Path, root: Path) -> None:
    """Aggregate file contents into arena.txt with clear headers."""
    with output_file.open("w", encoding="utf-8") as out:
        for path in paths:
            if not path.is_file():
                out.write(f"# --- ERROR: Not a file: {path} ---\n\n")
                continue
            try:
                content = path.read_text(encoding="utf-8")
                rel = path.relative_to(root).as_posix() if root else path.name
                out.write(f"# --- FILE: {rel} ---\n")
                out.write(content)
                if not content.endswith("\n"):
                    out.write("\n")
                out.write("\n")
            except Exception as exc:  # noqa: BLE001
                out.write(f"# --- ERROR reading {path}: {exc} ---\n\n")
```
